* このファイルについて

このファイルは、Emacs設定の原本となるLiterate Configurationです。

- 編集するファイル: =init.org=
- 自動生成されるファイル: =init.el=
- =init.el= は直接編集しない
- 設定は役割ごとの見出しに追加する

初回のみ、次のコマンドで =init.el= を生成します。

#+begin_example
M-x org-babel-tangle
#+end_example

以降は、このファイルを保存すると自動的に =init.el= を生成します。

** パッケージ管理とuse-package

Emacs 29以降では =use-package= が標準搭載されています。
MELPAをパッケージ取得先へ追加し、=use-package= の自動インストール機能を有効にします。
#+begin_src emacs-lisp
(require 'package)

;; MELPAで配布されているパッケージを利用する。
(add-to-list 'package-archives
             '("melpa" . "https://melpa.org/packages/")
             t)

(require 'use-package)
(require 'use-package-ensure)
#+end_src
*** use-packageの追加テンプレート

今後、パッケージの設定は基本的に次の形で追加します。

#+begin_example
(use-package package-name
  :ensure t
  :init
  ;; パッケージが読み込まれる前の設定
  :custom
  ;; Customize変数の設定
  :bind
  ;; キーバインド
  :hook
  ;; フック
  :config
  ;; パッケージが読み込まれた後の設定
  )
#+end_example

** Emacs本体の基本設定

Emacs標準の基本動作、ファイル保護、表示、編集支援をまとめて設定します。

=emacs= というパッケージを読み込むわけではなく、
Emacs本体の設定を =use-package= 形式で整理するために使用しています。
#+begin_src emacs-lisp
  (use-package emacs
    :ensure nil

    :init
    ;; カーソルを幅2pxの縦ラインにする。
    (setq-default cursor-type '(bar . 2))
    
    ;; Customizeが生成した設定をinit.elへ直接書き込ませない。
    ;; init.elはinit.orgから再生成されるため、別ファイルへ分離する。
    (setq custom-file
          (expand-file-name "custom.el" user-emacs-directory))

    (load custom-file t t)

    ;; バックアップファイルと自動保存ファイルを専用ディレクトリへ置く。
    (let ((backup-directory
           (expand-file-name "backups/" user-emacs-directory))
          (auto-save-directory
           (expand-file-name "auto-save/" user-emacs-directory)))

      (make-directory backup-directory t)
      (make-directory auto-save-directory t)

      (setq backup-directory-alist
            `(("." . ,backup-directory))

            auto-save-file-name-transforms
            `((".*" ,auto-save-directory t))))

    :custom
    ;; 起動画面を表示しない。
    (inhibit-startup-screen t)

    ;; *scratch*バッファの初期メッセージを表示しない。
    (initial-scratch-message nil)

    ;; yes/noではなくy/nで答えられるようにする。
    (use-short-answers t)

    ;; ビープ音を鳴らさない。
    (ring-bell-function #'ignore)

    ;; ファイル末尾に改行を入れる。
    (require-final-newline t)

    ;; カーソル移動によるスクロールを1行ずつにする。
    (scroll-conservatively most-positive-fixnum)

    ;; インデント済みの位置でTABを押した場合は補完を実行する。
    (tab-always-indent 'complete)

    :config
    ;; 選択範囲がある状態で入力すると、選択範囲を置換する。
    (delete-selection-mode 1)

    ;; 括弧を自動的に補完する。
    (electric-pair-mode 1)

    ;; 対応する括弧を強調表示する。
    (show-paren-mode 1)

    ;; 行番号と列番号を表示する。
    (global-display-line-numbers-mode 1)
    (column-number-mode 1)

    ;; GUIのツールバーとスクロールバーを非表示にする。
    ;; メニューバーはEmacsを覚えるのに役立つため残す。
    (when (fboundp 'tool-bar-mode)
      (tool-bar-mode -1))

    (when (fboundp 'scroll-bar-mode)
      (scroll-bar-mode -1)))
#+end_src
* macOSとの連携

macOS版Emacs固有の修飾キーと、GUI起動時のシェル環境を設定します。

** 修飾キー

OptionキーをEmacsのMetaキーとして使用します。
CommandキーはSuperキーとして使用し、macOS風のショートカットへ割り当てます。
#+begin_src emacs-lisp
(when (eq window-system 'mac)
  ;; OptionをMetaとして使う。
  (setq mac-option-modifier 'meta)

  ;; Commandを独立したSuper修飾キーとして使う。
  (setq mac-command-modifier 'super))
#+end_src

(use-package exec-path-from-shell

** Dired：標準のファイル管理
#+begin_src emacs-lisp
  (defface my-dired-file-face
    '((t (:inherit default)))
    "Face used for regular files in Dired."
    :group 'dired-faces)

  (defun my-dired-fontify-regular-files ()
    "Apply `my-dired-file-face' to regular files."
    (font-lock-add-keywords
     nil
     (list
      (list
       ;; 権限表示の先頭が「-」なら通常ファイル。
       (concat dired-re-maybe-mark
               dired-re-inode-size
               "-")
       '(".+" (dired-move-to-filename) nil
         (0 'my-dired-file-face))))
     'append))

  (require 'dired)
  (require 'project)

  (defun my-project-dired-other-window ()
    "プロジェクトルートを、一時的なDiredとして別ウィンドウに開く。"
    (interactive)
    (let* ((origin-window (selected-window))
           ;; Diredを開く前のウィンドウ構成を保存する。
           (window-config (current-window-configuration))
           (project       (project-current t))
           (root          (project-root project)))

      ;; ウィンドウを分割、または既存の別ウィンドウを使ってDiredを開く。
      (dired-other-window root)

      ;; Diredを表示しているウィンドウに、
      ;; 戻り先と元のウィンドウ構成を記録する。
      (set-window-parameter
       (selected-window)
       'my-dired-origin-window
       origin-window)

      (set-window-parameter
       (selected-window)
       'my-dired-window-config
       window-config)))

  (defun my-dired--clear-picker-state (window)
    "WINDOWに保存した一時Dired用の情報を削除する。"
    (set-window-parameter window 'my-dired-origin-window nil)
    (set-window-parameter window 'my-dired-window-config nil))

  (defun my-dired-find-file-and-restore ()
    "一時Diredなら、選択したファイルを元のウィンドウで開いてDiredを終了する。"
    (interactive)
    (let* ((dired-window  (selected-window))
           (origin-window
            (window-parameter dired-window
                              'my-dired-origin-window))
           (window-config
            (window-parameter dired-window
                              'my-dired-window-config))
           (file (dired-get-file-for-visit)))

      ;; 通常の方法で開いたDiredでは、標準動作のままにする。
      (if (not (window-configuration-p window-config))
          (dired-find-file)

        ;; ディレクトリを選んだ場合は、Dired内で移動を続ける。
        (if (file-directory-p file)
            (dired-find-alternate-file)

          ;; ファイルは、まず表示せずにバッファへ読み込む。
          ;; 読み込みに失敗した場合はDiredを閉じない。
          (let ((file-buffer  (find-file-noselect file))
                (dired-buffer (current-buffer)))

            ;; 一時Diredの状態を解除する。
            (my-dired--clear-picker-state dired-window)

            ;; Diredを開く前のウィンドウ構成へ戻す。
            (set-window-configuration window-config)

            ;; 元の作業ウィンドウを選択する。
            (when (window-live-p origin-window)
              (select-window origin-window))

            ;; 選択したファイルを元の作業ウィンドウに表示する。
            (switch-to-buffer file-buffer)

            ;; 一時的に使ったDiredバッファを削除する。
            (when (buffer-live-p dired-buffer)
              (kill-buffer dired-buffer)))))))

  (defun my-dired-quit-and-restore ()
    "一時Diredをキャンセルし、元のウィンドウ構成へ戻す。"
    (interactive)
    (let* ((dired-window  (selected-window))
           (origin-window
            (window-parameter dired-window
                              'my-dired-origin-window))
           (window-config
            (window-parameter dired-window
                              'my-dired-window-config)))

      ;; 通常のDiredなら、標準の終了動作を行う。
      (if (not (window-configuration-p window-config))
          (quit-window)

        (let ((dired-buffer (current-buffer)))
          (my-dired--clear-picker-state dired-window)
          (set-window-configuration window-config)

          (when (window-live-p origin-window)
            (select-window origin-window))

          (when (buffer-live-p dired-buffer)
            (kill-buffer dired-buffer))))))

  (defun my-dired-disable-line-numbers ()
    "Diredでは行番号を表示しない。"
    (display-line-numbers-mode -1))

  (use-package dired
    :ensure nil

    :hook
    ;; 所有者、権限、日時などを最初は隠して、一覧をすっきり表示する。
    ((dired-mode . dired-hide-details-mode)
     (dired-mode . my-dired-fontify-regular-files)
     (dired-mode . my-dired-disable-line-numbers))

    :bind
    ("C-c d" . my-project-dired-other-window)

    :config
    ;; RET、f、eなど、`dired-find-file'を呼ぶキーをまとめて置き換える。
    (define-key dired-mode-map
                [remap dired-find-file]
                #'my-dired-find-file-and-restore)

    ;; qでキャンセルした場合も、元の画面へ完全に戻す。
    (define-key dired-mode-map
                (kbd "q")
                #'my-dired-quit-and-restore)

    :custom
    ;; 2つのDiredを左右に表示している場合、
    ;; コピー・移動先の初期値をもう一方のディレクトリにする。
    (dired-dwim-target t)

    ;; ディレクトリを移動してもDiredバッファを大量に残さない。
    (dired-kill-when-opening-new-dired-buffer t)

    ;; ディレクトリの再帰コピーは確認なしで行う。
    (dired-recursive-copies 'always)

    ;; 再帰削除では最上位で一度確認する。
    (dired-recursive-deletes 'top)

    ;; 削除したファイルを直接消さず、macOSのゴミ箱へ移動する。
    (delete-by-moving-to-trash t)

    ;; 隠しファイルは表示するが、`.` と `..` は表示しない。
    ;(dired-listing-switches "-Al")

    :custom-face
    ;; 通常ファイル
    (my-dired-file-face
     ((t (:inherit font-lock-string-face))))

    ;; ディレクトリ
    (dired-directory
     ((t (:inherit font-lock-type-face
                   :weight bold)))))
#+end_src

** Tab Bar：作業画面全体の切り替え

画面分割を含む作業画面全体を、タブとして切り替えます。

| キー       | 操作                |
|------------+---------------------|
| =C-PageDown= | 右隣のTab Barへ移動 |
| =C-PageUp=   | 左隣のTab Barへ移動 |

どちらも端まで移動すると、反対側へ戻って循環します。
#+begin_src emacs-lisp
  (use-package tab-bar
    :ensure nil

    :custom
    ;; 新しいTab Barは、ほかのタブと共有しない専用Scratchで始める。
    (tab-bar-new-tab-choice
     #'bufferlo-create-local-scratch-buffer)

    ;; 新しいタブを作る「＋」ボタンを表示しない。
    (tab-bar-new-button-show nil)

    ;; 各タブの「×」ボタンを表示しない。
    (tab-bar-close-button-show nil)

    :config
    ;; Emacs標準のTab Barを有効にする。
    (tab-bar-mode 1)

    ;; Ctrl+PageDown：右隣のTab Barへ移動する。
    ;; 右端では左端へ戻り、循環する。
    (keymap-global-set "C-<next>" #'tab-next)

    ;; Ctrl+PageUp：左隣のTab Barへ移動する。
    ;; 左端では右端へ戻り、循環する。
    (keymap-global-set "C-<prior>" #'tab-previous))
#+end_src

** Bufferlo：Tab Barごとのバッファ管理

各Tab Barが独立したバッファ一覧を持つようにします。

#+begin_src emacs-lisp
  (use-package bufferlo
    :ensure t
    :demand t

    :init
    ;; next-bufferとprevious-bufferも現在のTab Bar内に限定する。
    ;; bufferlo-modeを有効にする前に設定する必要がある。
    (setq bufferlo-prefer-local-buffers 'tabs)

    :bind
    ;; 現在のTab Barに属するバッファから選択する。
    ;; Prefix引数を付けると、Emacs全体のバッファから選択できる。
    (("C-x b" . bufferlo-switch-to-buffer))

    :config
    ;; フレーム・Tab Barごとのローカルバッファ管理を有効にする。
    (bufferlo-mode 1))
#+end_src
** Tab Line：ウィンドウ内のバッファ切り替え

各ウィンドウで表示したバッファを、タブとして切り替えます。

| キー         | 操作                 |
|--------------+----------------------|
| =Cmd+Option+←= | 左隣のTab Lineへ移動 |
| =Cmd+Option+→= | 右隣のTab Lineへ移動 |

どちらも端まで移動すると、反対側へ戻って循環します。

Tab Lineのバッファ名は並び順を維持し、
新しく表示したバッファだけを右端へ追加します。
#+begin_src emacs-lisp
  (use-package tab-line
    :ensure nil
    :after bufferlo

    :init
    (require 'seq)

    ;; Tab Lineに表示しないバッファ名。
    (defconst my/tab-line-excluded-buffer-names
      '("*scratch*")
      "Tab Lineに表示しないバッファ名の一覧。")

    (defun my/tab-line-buffer-visible-p (buffer)
      "BUFFERをTab Lineに表示するとき、非nilを返す。"
      (and
       ;; 削除済みのバッファは表示しない。
       (buffer-live-p buffer)

       ;; 通常のScratchと、Bufferloが作るタブ専用Scratchを表示しない。
       (not
        (string-prefix-p
         "*scratch*"
         (buffer-name buffer)))

       ;; 表示しないバッファはここに定義する。
       (not
        (with-current-buffer buffer
          (derived-mode-p
           'dired-mode
           'magit-mode
           'vterm-mode)))))

    (defun my/tab-line-tabs ()
      "不要なバッファを除外し、固定順でTab Lineへ表示する。"
      (let* (;; このウィンドウで保存している、前回までの並び順。
             (old-buffers
              (seq-filter
               #'my/tab-line-buffer-visible-p
               (window-parameter nil 'tab-line-buffers)))

             ;; 現在、このウィンドウに関連付いているバッファ。
             (current-buffers
              (seq-filter
               #'my/tab-line-buffer-visible-p
               (bufferlo-buffer-list)))

             ;; 以前からあるバッファは、保存済みの順番を維持する。
             (kept-buffers
              (seq-filter
               (lambda (buffer)
                 (memq buffer current-buffers))
               old-buffers))

             ;; 新しく表示されたバッファは、右端へ追加する。
             (added-buffers
              (seq-remove
               (lambda (buffer)
                 (memq buffer kept-buffers))
               current-buffers))

             (buffers
              (append kept-buffers added-buffers)))

        ;; ウィンドウごとのTab Lineの並び順として保存する。
        (set-window-parameter nil 'tab-line-buffers buffers)

        buffers))

    (defun my/close-tab-or-window ()
      "Buffer、Window、Tab Barの順で閉じる。
  最後のTab BarではEmacsを終了しない。"
      (interactive)
      (let ((visible-buffer-count
             ;; 現在のTab BarでTab Lineへ表示されるバッファ数。
             (length
              (funcall tab-line-tabs-function)))
            (tab-count
             ;; 現在のフレームに存在するTab Barの数。
             (length
              (funcall tab-bar-tabs-function))))
        (cond
         ;; ウィンドウが複数ある場合は、従来の挙動を維持する。
         ;;
         ;; 表示対象バッファがあれば、このTab Barからだけ外す。
         ;; 表示対象バッファがなければ、現在のウィンドウを閉じる。
         ((not (one-window-p))
          (if (> visible-buffer-count 0)
              (bufferlo-bury)
            (delete-window)))

         ;; Tab Barが複数ある場合：
         ;; 他のTab Barで同じバッファを使っている可能性があるため、
         ;; バッファ自体は削除せず、このTab Barからだけ外す。
         ((> tab-count 1)
          (if (> visible-buffer-count 0)
              (bufferlo-bury)
            (tab-close)))

         ;; ここからは、最後のウィンドウかつ最後のTab Bar。
         ;;
         ;; Scratch以外ならバッファ自体を削除する。
         ;; Dired、Magit、vtermもここで削除できる。
         ((not
           (string-prefix-p
            "*scratch*"
            (buffer-name)))
          (kill-current-buffer))

         ;; Scratchだけが表示されている状態を、
         ;; VS Codeの「何も開かれていない状態」とみなす。
         (t
          (message "最後のTab Barなので、Emacsは終了しません")))))

    ;; Tab Barを切り替えても、ウィンドウごとの並び順を保持する。
    (add-to-list
     'window-persistent-parameters
     '(tab-line-buffers . t))

    :custom
    ;; Tab Lineへ表示するバッファを独自に絞り込む。
    (tab-line-tabs-function #'my/tab-line-tabs)

    ;; 右端・左端へ到達したら、反対側へ戻って循環する。
    (tab-line-switch-cycling t)

    ;; 各タブの閉じるボタンを表示しない。
    (tab-line-close-button-show nil)

    :config
    ;; すべてのウィンドウでTab Lineを有効にする。
    (global-tab-line-mode 1)

    ;; Cmd+Option+←：左隣のTab Lineへ移動する。
    (keymap-global-set
     "M-s-<left>"
     #'tab-line-switch-to-prev-tab)

    ;; Cmd+Option+→：右隣のTab Lineへ移動する。
    (keymap-global-set
     "M-s-<right>"
     #'tab-line-switch-to-next-tab))

** 作業項目の移動

=inbox.org= の項目上で =C-c w= を押すと、
現在項目を =cockpit.org= の =Work Items= へ移動します。

本文や子見出しにカーソルがある場合も、Inbox直下の項目全体を移動します。
#+begin_src emacs-lisp
(defun my/org-refile-location (file headline)
  "FILE内のHEADLINEを、`org-refile'用の移動先として返す。"
  (let* ((buffer (find-file-noselect file))
         (target-file (buffer-file-name buffer))
         (position
          (with-current-buffer buffer
            (org-with-wide-buffer
              (goto-char (point-min))
              (or (org-find-exact-headline-in-buffer headline)
                  (user-error
                   "%sに見出し「%s」がありません"
                   file
                   headline))))))
    (list headline target-file nil position)))

(defun my/save-task-org-buffers ()
  "変更されたInboxとCockpitのバッファを保存する。"
  (dolist (file (list my/task-inbox-file
                      my/task-cockpit-file))
    (let ((buffer (find-buffer-visiting file)))
      (when buffer
        (with-current-buffer buffer
          (when (buffer-modified-p)
            (save-buffer)))))))

(defun my/move-inbox-item-to-work-items ()
  "Inboxの現在項目をCockpitのWork Itemsへ移動する。"
  (interactive)
  (unless (and buffer-file-name
               (file-equal-p buffer-file-name
                             my/task-inbox-file))
    (user-error "このコマンドはinbox.orgでのみ使用できます"))

  (org-back-to-heading t)

  ;; 子見出し内から実行した場合も、Inbox直下の項目全体を対象にする。
  (while (> (org-outline-level) 2)
    (org-up-heading-safe))

  (unless (and (= (org-outline-level) 2)
               (save-excursion
                 (org-up-heading-safe)
                 (equal (org-get-heading t t t t)
                        "Inbox")))
    (user-error "Inbox直下の項目で実行してください"))

  (org-refile
   nil
   nil
   (my/org-refile-location
    my/task-cockpit-file
    "Work Items"))

  (my/save-task-org-buffers)
  (message "Inbox項目をCockpitのWork Itemsへ移動しました"))

(with-eval-after-load 'org
  (keymap-set
   org-mode-map
   "C-c w"
   #'my/move-inbox-item-to-work-items))

** 作業状態の一覧表示

*** Cockpit Agenda

=C-c a c= で、Cockpitの作業状態を一覧表示します。
表示上の分類はTODO状態から自動生成します。

| 表示      | 対象状態            |
|-----------+---------------------|
| Active    | =DOING=               |
| Next      | =NEXT=                |
| Suspended | =HOLD=                |
| Waiting   | =WAIT=                |
| Planned   | =TODO=のCockpit項目 |
| Inbox     | =TODO=のInbox項目   |

Agenda項目上では、次のキーで表示順に隣接するレーンへ移動します。
移動後は同じ項目へカーソルも追従するため、
Cmdを押したまま上下を連続入力できます。

| キー      | 操作                              |
|-----------+-----------------------------------|
| =Cmd+↑=     | 1つ上のワークフローレーンへ移動 |
| =Cmd+↓=     | 1つ下のワークフローレーンへ移動 |
| =Option+↑=  | Agenda標準の行移動               |
| =Option+↓=  | Agenda標準の行移動               |

Cockpit内の移動ではTODO状態を変更し、
PlannedとInboxの境界では項目をファイル間で移動します。
#+begin_src emacs-lisp
(require 'cl-lib)

(defconst my/org-agenda-workflow-lanes
  `((:name "Active"
     :file ,my/task-cockpit-file
     :todo "DOING"
     :parent "Work Items")
    (:name "Next"
     :file ,my/task-cockpit-file
     :todo "NEXT"
     :parent "Work Items")
    (:name "Suspended"
     :file ,my/task-cockpit-file
     :todo "HOLD"
     :parent "Work Items")
    (:name "Waiting"
     :file ,my/task-cockpit-file
     :todo "WAIT"
     :parent "Work Items")
    (:name "Planned"
     :file ,my/task-cockpit-file
     :todo "TODO"
     :parent "Work Items")
    (:name "Inbox"
     :file ,my/task-inbox-file
     :todo "TODO"
     :parent "Inbox"))
  "Cockpit Agendaで上下移動するワークフローレーン。")

(defun my/org-agenda-entry-marker ()
  "現在のAgenda行が参照するOrg見出しのマーカーを返す。"
  (or (org-get-at-bol 'org-hd-marker)
      (org-get-at-bol 'org-marker)
      (user-error "Agenda項目の行にカーソルを置いてください")))

(defun my/org-last-refile-marker ()
  "直前にRefileされたOrg見出しを指す新しいマーカーを返す。"
  (require 'bookmark)
  (let* ((bookmark-name
          (plist-get org-bookmark-names-plist
                     :last-refile))
         (bookmark
          (and bookmark-name
               (bookmark-get-bookmark
                bookmark-name
                'noerror)))
         (file
          (and bookmark
               (bookmark-get-filename bookmark)))
         (position
          (and bookmark
               (bookmark-get-position bookmark))))
    (unless (and file position)
      (user-error "Refile後の項目位置を取得できません"))
    (with-current-buffer (find-file-noselect file)
      (copy-marker position))))

(defun my/org-agenda-goto-marker (marker column)
  "Agenda内でMARKERに対応する項目へ移動し、COLUMNを復元する。"
  (goto-char (point-min))
  (let (found)
    (while (and (not found)
                (not (eobp)))
      (let ((line-marker
             (or (org-get-at-bol 'org-hd-marker)
                 (org-get-at-bol 'org-marker))))
        (if (and line-marker
                 (equal line-marker marker))
            (setq found t)
          (forward-line 1))))

    (unless found
      (user-error "移動後のAgenda項目を見つけられません"))

    (org-move-to-column column)
    (org-agenda-do-context-action)))

(defun my/org-agenda-lane-index (file todo)
  "FILEとTODOに対応するワークフローレーンの位置を返す。"
  (or
   (cl-position-if
    (lambda (lane)
      (and (file-equal-p file
                         (plist-get lane :file))
           (equal todo
                  (plist-get lane :todo))))
    my/org-agenda-workflow-lanes)
   (user-error
    "現在の項目はレーン移動の対象外です: %s / %s"
    file
    todo)))

(defun my/org-agenda-move-workflow-lane (offset)
  "Agenda項目とカーソルをOFFSET方向のワークフローレーンへ移動する。"
  (let* ((column (current-column))
         (marker (my/org-agenda-entry-marker))
         (source-file
          (org-with-point-at marker
            (buffer-file-name)))
         (source-todo
          (org-with-point-at marker
            (org-get-todo-state)))
         (source-index
          (my/org-agenda-lane-index
           source-file
           source-todo))
         (target-index (+ source-index offset)))

    (unless (< -1
               target-index
               (length my/org-agenda-workflow-lanes))
      (user-error
       "これ以上%sのレーンへは移動できません"
       (if (< offset 0) "上" "下")))

    (let* ((source-lane
            (nth source-index
                 my/org-agenda-workflow-lanes))
           (target-lane
            (nth target-index
                 my/org-agenda-workflow-lanes))
           (target-file
            (plist-get target-lane :file))
           (target-todo
            (plist-get target-lane :todo))
           moved-marker)

      (unwind-protect
          (progn
            (if (file-equal-p source-file target-file)
                ;; Cockpit内では、元見出しのTODO状態だけを変更する。
                (progn
                  (org-agenda-todo target-todo)
                  (setq moved-marker
                        (copy-marker
                         org-last-heading-marker)))

              ;; InboxとCockpitの境界では、見出し全体をファイル間移動する。
              (org-agenda-refile
               nil
               (my/org-refile-location
                target-file
                (plist-get target-lane :parent))
               t)
              (setq moved-marker
                    (my/org-last-refile-marker)))

            (my/save-task-org-buffers)
            (org-agenda-redo)
            (my/org-agenda-goto-marker
             moved-marker
             column)

            (message
             "%sから%sへ移動しました"
             (plist-get source-lane :name)
             (plist-get target-lane :name)))

        (when (markerp moved-marker)
          (set-marker moved-marker nil))))))

(defun my/org-agenda-move-workflow-up ()
  "Agenda項目を1つ上のワークフローレーンへ移動する。"
  (interactive)
  (my/org-agenda-move-workflow-lane -1))

(defun my/org-agenda-move-workflow-down ()
  "Agenda項目を1つ下のワークフローレーンへ移動する。"
  (interactive)
  (my/org-agenda-move-workflow-lane 1))

(use-package org-agenda
  :ensure nil
  :after org

  :bind
  (:map org-agenda-mode-map
        ("s-<up>" . my/org-agenda-move-workflow-up)
        ("s-<down>" . my/org-agenda-move-workflow-down))

  :config
  (setq org-agenda-custom-commands
        `(("c" "Cockpit"
           (
            ;; 現在作業中。
            (todo "DOING"
                  ((org-agenda-overriding-header "Active")
                   (org-agenda-files
                    '(,my/task-cockpit-file))))

            ;; 次に着手できる作業。
            (todo "NEXT"
                  ((org-agenda-overriding-header "Next")
                   (org-agenda-files
                    '(,my/task-cockpit-file))))

            ;; 中断中の作業。
            (todo "HOLD"
                  ((org-agenda-overriding-header "Suspended")
                   (org-agenda-files
                    '(,my/task-cockpit-file))))

            ;; 外部条件や回答待ち。
            (todo "WAIT"
                  ((org-agenda-overriding-header "Waiting")
                   (org-agenda-files
                    '(,my/task-cockpit-file))))

            ;; Cockpitへ登録済みだが、まだ着手順を決めていない作業。

*** Vterm Toggle

Cmd+jでプロジェクト単位のvtermを全画面表示・非表示にします。
Cmd+Shift+jでは、現在のウィンドウ構成に応じて分割表示します。

- 1ウィンドウ、または上下分割: フレームの右50%
- 左右分割: フレームの下30%

#+begin_src emacs-lisp
(use-package vterm-toggle
  :ensure t

  :preface
  (defun my/vterm-toggle--left-right-layout-p ()
    "選択ウィンドウに左右隣接するウィンドウがあるとき、非nilを返す。"
    (or (window-in-direction 'left)
        (window-in-direction 'right)))

  (defun my/vterm-toggle--display-buffer (buffer _alist)
    "現在のウィンドウ構成に応じたフレーム位置へBUFFERを表示する。"
    (if (my/vterm-toggle--left-right-layout-p)
        (display-buffer-in-direction
         buffer
         '((direction . below)
           (window . root)
           (window-height . 0.30)))
      (display-buffer-in-direction
       buffer
       '((direction . right)
         (window . root)
         (window-width . 0.50)))))

  (defun my/vterm-toggle-split ()
    "現在のウィンドウ構成に応じてvtermを分割表示・非表示にする。"
    (interactive)
    (let ((vterm-toggle-fullscreen-p nil)
          (display-buffer-alist
           (cons
            '((lambda (buffer-or-name _action)
                (let ((buffer (get-buffer buffer-or-name)))
                  (and buffer
                       (with-current-buffer buffer
                         (derived-mode-p 'vterm-mode)))))
              (display-buffer-reuse-window
               my/vterm-toggle--display-buffer))
            display-buffer-alist)))
      (vterm-toggle)))

  ;; vterm-toggleを遅延読み込み可能なコマンドとして登録する。
  :commands vterm-toggle

  ;; Emacs起動直後からCmd+jとCmd+Shift+jを使えるようにする。
  :bind
  (("s-j" . vterm-toggle)
   ("s-J" . my/vterm-toggle-split))

  :custom
  ;; 通常のvterm-toggleは全画面で表示する。
  (vterm-toggle-fullscreen-p t)

  ;; vtermを隠したとき、元のウィンドウ構成へ戻す。
  ;; configrationはパッケージ側の変数名どおりの綴り。
  (vterm-toggle-hide-method 'reset-window-configration)

  ;; プロジェクトごとにvtermを管理する。
  (vterm-toggle-scope 'project)

  ;; 新しいvtermをプロジェクトルートで起動する。
  (vterm-toggle-project-root t))

  (defun my/apply-default-font (&optional frame)
    "FRAMEへ通常バッファ用の標準フォントを設定する。"
    (with-selected-frame (or frame (selected-frame))
      (when (display-graphic-p)

    "現在のOrgバッファへUDEV Gothic NFを適用する。"

     'default
     :family "UDEV Gothic NF")

    (face-remap-add-relative

     :family "UDEV Gothic NF")

     'variable-pitch
     :family "UDEV Gothic NF"))

  (setq-default line-spacing 0.25)

  (add-hook

** 独自の編集コマンド

*** 現在の行を選択
#+begin_src emacs-lisp
(defun my/select-current-line ()
  "カーソル行全体を選択し、カーソルを次の行頭へ移動する。
 続けて実行すると、選択範囲を次の行まで広げる。"
  (interactive)
  (if (and (region-active-p)
           (eq last-command 'my/select-current-line))
      ;; 連続実行時は、選択範囲を次の行まで広げる。
      (goto-char (line-beginning-position 2))

    ;; 最初の実行では、現在行の先頭から次の行頭までを選択する。
    (let ((start (line-beginning-position)))
      (goto-char (line-beginning-position 2))
      (set-mark start)
      (activate-mark))))

;; VS Code風にカーソル行を選択する。
(keymap-global-set "s-l" #'my/select-current-line)
#+end_src

*** カーソル行の上下に新しい行を挿入

VS Codeの「Insert Line Below」「Insert Line Above」に相当する操作です。
カーソルが行の途中にあっても、現在行の内容は分割しません。

- =Cmd+Enter=: カーソル行の下に新しい行を追加し、その行へ移動する
- =Cmd+Shift+Enter=: カーソル行の上に新しい行を追加し、その行へ移動する

#+begin_src emacs-lisp
(defun my/insert-line-below ()
  "カーソル行の下に新しい行を挿入し、その行へ移動する。"
  (interactive)
  (end-of-line)
  (newline-and-indent))

(defun my/insert-line-above ()
  "カーソル行の上に新しい行を挿入し、その行へ移動する。"
  (interactive)
  (beginning-of-line)
  (open-line 1)
  (indent-according-to-mode))

;; VS Code風に、現在行の下へ新しい行を挿入する。
(keymap-global-set "s-<return>" #'my/insert-line-below)

;; VS Code風に、現在行の上へ新しい行を挿入する。
(keymap-global-set "S-s-<return>" #'my/insert-line-above)
#+end_src

*** macOS風の基本操作

保存、選択、クリップボード、Undo、カーソル移動をmacOS風のキーへ割り当てます。
#+begin_src emacs-lisp
  ;; macOS風の保存。
  (keymap-global-set "s-s" #'save-buffer)

  ;; macOS風の全選択。
  (keymap-global-set "s-a" #'mark-whole-buffer)

  ;; macOS風のコピー。
  (keymap-global-set "s-c" #'kill-ring-save)

  (keymap-global-set "s-v" #'yank)

  ;; Tab Lineの最後のタブならウィンドウを閉じ、
  ;; それ以外なら現在のタブを閉じる。
  (keymap-global-set "s-w" #'my/close-tab-or-window)

  ;; Cmd+←：行頭へ移動する。
  (keymap-global-set "s-<left>" #'move-beginning-of-line)

  ;; Cmd+→：行末へ移動する。
  (keymap-global-set "s-<right>" #'move-end-of-line)

  (keymap-global-set "s-<up>" #'beginning-of-buffer)

  ;; Cmd+↓：バッファの末尾へ移動する。
  (keymap-global-set "s-<down>" #'end-of-buffer)

    (interactive)
    (delete-region (line-beginning-position) (point)))

    (interactive)
    (delete-region (point) (line-end-position)))

  ;; カーソル位置から行頭までを削除する。

                  #'my/delete-to-line-beginning)

  ;; カーソル位置から行末までを削除する。
  (global-set-key (kbd "s-<delete>")
                  #'my/delete-to-line-end)
